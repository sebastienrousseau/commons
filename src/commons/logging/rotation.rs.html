5//! Log rotation policies: size, time, date, and count-based.
6//!
7//! Wrap a file sink with [`RotatingFile`] to enforce automatic rotation.
8//! On rotation, the current file is renamed with a timestamp suffix and
9//! a fresh file is opened at the original path.
10
11use super::log_config::LogRotation;
12use std::fs::{self, File, OpenOptions};
13use std::io::{self, Write};
14use std::path::{Path, PathBuf};
15use std::time::Instant;
16
17/// File writer that enforces a [`LogRotation`] policy.
18#[derive(Debug)]
19pub struct RotatingFile {
20    /// Current open file handle.
21    file: File,
22    /// Path to the current log file.
23    path: PathBuf,
24    /// Rotation policy to enforce.
25    policy: LogRotation,
26    /// Bytes written to the current file.
27    bytes_written: u64,
28    /// Events written to the current file (for count-based rotation).
29    events_written: u32,
30    /// Time when the current file was opened (for time-based rotation).
31    opened_at: Instant,
32    /// Date string when the current file was opened (for date-based rotation).
33    opened_date: String,
34}
35
36impl RotatingFile {
37    /// Open (or create) a log file with the given rotation policy.
38    ///
39    /// # Errors
40    ///
41    /// Returns `io::Error` if the file cannot be opened or created.
42    pub fn open(path: &Path, policy: LogRotation) -> io::Result<Self> {
43        let file = OpenOptions::new().create(true).append(true).open(path)?;
44        let bytes_written = file.metadata().map(|m| m.len()).unwrap_or(0);
45        Ok(Self {
46            file,
47            path: path.to_path_buf(),
48            policy,
49            bytes_written,
50            events_written: 0,
51            opened_at: Instant::now(),
52            opened_date: today_date_string(),
53        })
54    }
55
56    /// Write a batch of bytes, then rotate if the policy threshold is met.
57    ///
58    /// # Errors
59    ///
60    /// Returns `io::Error` if the write or file rotation fails.
61    pub fn write_batch(&mut self, data: &[u8], event_count: u32) -> io::Result<()> {
62        self.file.write_all(data)?;
63        self.bytes_written += data.len() as u64;
64        self.events_written += event_count;
65
66        if self.should_rotate() {
67            self.rotate()?;
68        }
69        Ok(())
70    }
71
72    /// Checks whether the current file should be rotated.
73    fn should_rotate(&self) -> bool {
74        match self.policy {
75            LogRotation::Size(max_bytes) => self.bytes_written >= max_bytes.get(),
76            LogRotation::Time(seconds) => self.opened_at.elapsed().as_secs() >= seconds.get(),
77            LogRotation::Date => today_date_string() != self.opened_date,
78            LogRotation::Count(max_events) => self.events_written >= max_events,
79        }
80    }
81
82    /// Rotates the current file by renaming it with a timestamp suffix
83    /// and opening a new file at the original path.
84    fn rotate(&mut self) -> io::Result<()> {
85        // Flush and drop the current file handle.
86        self.file.flush()?;
87
88        // Build the rotated file name.
89        let timestamp = chrono_like_timestamp();
90        let rotated_name = if let Some(ext) = self.path.extension() {
91            let stem = self.path.with_extension("");
92            PathBuf::from(format!(
93                "{}.{timestamp}.{}",
94                stem.display(),
95                ext.to_string_lossy()
96            ))
97        } else {
98            PathBuf::from(format!("{}.{timestamp}", self.path.display()))
99        };
100
101        fs::rename(&self.path, &rotated_name)?;
102
103        // Open a new file at the original path.
104        self.file = OpenOptions::new()
105            .create(true)
106            .append(true)
107            .open(&self.path)?;
108        self.bytes_written = 0;
109        self.events_written = 0;
110        self.opened_at = Instant::now();
111        self.opened_date = today_date_string();
112
113        Ok(())
114    }
115}

117/// Returns today's date as `YYYY-MM-DD`.
118fn today_date_string() -> String {
119    let now = std::time::SystemTime::now()
120        .duration_since(std::time::UNIX_EPOCH)
121        .unwrap_or_default();
122    let secs = now.as_secs();
123    // Simple date calculation (no leap-second precision needed for rotation).
124    let days = secs / 86400;
125    let (year, month, day) = days_to_ymd(days);
126    format!("{year:04}-{month:02}-{day:02}")
127}
128
129/// Returns a compact timestamp for rotated file names: `YYYYMMDD-HHMMSS`.
130fn chrono_like_timestamp() -> String {
131    let now = std::time::SystemTime::now()
132        .duration_since(std::time::UNIX_EPOCH)
133        .unwrap_or_default();
134    let secs = now.as_secs();
135    let days = secs / 86400;
136    let (year, month, day) = days_to_ymd(days);
137    let day_secs = secs % 86400;
138    let h = day_secs / 3600;
139    let m = (day_secs % 3600) / 60;
140    let s = day_secs % 60;
141    format!("{year:04}{month:02}{day:02}-{h:02}{m:02}{s:02}")
142}
143
144/// Converts days since Unix epoch to (year, month, day).
145const fn days_to_ymd(days: u64) -> (u64, u64, u64) {
146    // Algorithm from http://howardhinnant.github.io/date_algorithms.html
147    let z = days + 719_468;
148    let era = z / 146_097;
149    let doe = z - era * 146_097;
150    let yoe = (doe - doe / 1460 + doe / 36524 - doe / 146_096) / 365;
151    let y = yoe + era * 400;
152    let doy = doe - (365 * yoe + yoe / 4 - yoe / 100);
153    let mp = (5 * doy + 2) / 153;
154    let d = doy - (153 * mp + 2) / 5 + 1;
155    let m = if mp < 10 { mp + 3 } else { mp - 9 };
156    let y = if m <= 2 { y + 1 } else { y };
157    (y, m, d)
158}

165    #[test]
166    fn test_today_date_string_format() {
167        let date = today_date_string();
168        // YYYY-MM-DD
169        assert_eq!(date.len(), 10);
170        assert_eq!(&date[4..5], "-");
171        assert_eq!(&date[7..8], "-");

178        assert_eq!(ts.len(), 15);
179        assert_eq!(&ts[8..9], "-");
180    }

186    }

188    #[test]
189    fn test_rotating_file_size_based() {
190        let dir = tempfile::tempdir().unwrap();
191        let path = dir.path().join("test.log");
192        let policy = LogRotation::Size(NonZeroU64::new(100).unwrap());
193        let mut rf = RotatingFile::open(&path, policy).unwrap();
194        // Write 50 bytes -- no rotation
195        rf.write_batch(&[b'A'; 50], 1).unwrap();
196        assert!(path.exists());
197        // Write 60 more bytes -- triggers rotation
198        rf.write_batch(&[b'B'; 60], 1).unwrap();
199        // Original path should still exist (new file)
200        assert!(path.exists());
201        // There should be a rotated file
202        let entries: Vec<_> = fs::read_dir(dir.path())
203            .unwrap()
204            .filter_map(Result::ok)
205            .collect();
206        assert!(entries.len() >= 2, "expected rotated file");
207    }

248        let dir = tempfile::tempdir().unwrap();
249        // File without extension
250        let path = dir.path().join("logfile");
251        let policy = LogRotation::Size(NonZeroU64::new(10).unwrap());

271        assert!(
272            name_str.starts_with("logfile."),
273            "rotated file should be logfile.<timestamp>"
274        );

277    #[test]
278    fn test_days_to_ymd_known_dates() {
279        // 2024-01-01 is 19723 days after epoch (1970-01-01)
280        let (y, m, d) = days_to_ymd(19723);
281        assert_eq!((y, m, d), (2024, 1, 1));
282
283        // 2000-03-01: leap year edge (day 11017)
284        let (y, m, d) = days_to_ymd(11017);
285        assert_eq!((y, m, d), (2000, 3, 1));
286
287        // 2000-02-29: leap day (day 11016)
288        let (y, m, d) = days_to_ymd(11016);
289        assert_eq!((y, m, d), (2000, 2, 29));
290    }
