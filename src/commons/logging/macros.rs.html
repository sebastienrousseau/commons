5//! Convenience macros for span tracking, latency profiling, and MCP notifications.
6//!
7//! All macros dispatch through the lock-free [`ENGINE`](super::engine::ENGINE).
8
9/// Execute a block within an OTLP-tagged span.
10///
11/// Emits an OTLP log with a generated `span_id`, increments the active
12/// span counter, runs `$block`, then decrements. Returns the block's value.
13///
14/// Requires the `id` feature for span ID generation.
15#[cfg(feature = "id")]
16#[macro_export]
17macro_rules! logging_span {
18    ($name:expr, $block:block) => {{
19        let span_id = $crate::logging::utils::generate_span_id();
20        $crate::logging::engine::ENGINE.inc_spans();
21        $crate::logging::log_entry::Log::info($name)
22            .with("span_id", &span_id)
23            .format($crate::logging::log_format::LogFormat::OTLP)
24            .fire();
25        let result = $block;
26        $crate::logging::engine::ENGINE.dec_spans();
27        result
28    }};
29}

31/// Measure wall-clock latency of a block and emit a Logfmt metric.
32///
33/// Captures `Instant::now()` before the block, computes elapsed
34/// microseconds after, and fires a Logfmt log with `latency_us`.
35#[macro_export]
36macro_rules! logging_time_it {
37    ($action:expr, $block:block) => {{
38        let start = std::time::Instant::now();
39        let result = $block;
40        let elapsed = start.elapsed().as_micros();
41
42        $crate::logging::log_entry::Log::info(&format!("{} completed", $action))
43            .with("latency_us", elapsed as u64)
44            .format($crate::logging::log_format::LogFormat::Logfmt)
45            .fire();
46
47        result
48    }};
49}

51/// Emit an MCP-formatted state transition notification.
52///
53/// Use for AI agent orchestration where state changes must be
54/// machine-readable via JSON-RPC 2.0 notification semantics.
55#[macro_export]
56macro_rules! logging_mcp_notify {
57    ($state_key:expr, $state_val:expr) => {
58        $crate::logging::log_entry::Log::info("State transition")
59            .with($state_key, $state_val)
60            .format($crate::logging::log_format::LogFormat::MCP)
61            .fire();
62    };
63}
