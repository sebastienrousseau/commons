5//! Bridge from the [`log`](https://docs.rs/log) crate facade into the logging engine.
6//!
7//! Installed automatically by [`init()`](super::init::init) unless
8//! you call `.without_log()` on the builder.
9
10use super::engine::ENGINE;
11use super::log_entry::Log;
12use super::log_format::LogFormat;
13use super::log_level::LogLevel;
14
15/// Convert a [`log::Level`] to the corresponding [`LogLevel`].
16#[must_use]
17pub const fn map_log_level(level: log::Level) -> LogLevel {
18    match level {
19        log::Level::Error => LogLevel::ERROR,
20        log::Level::Warn => LogLevel::WARN,
21        log::Level::Info => LogLevel::INFO,
22        log::Level::Debug => LogLevel::DEBUG,
23        log::Level::Trace => LogLevel::TRACE,
24    }
25}
26
27/// Convert a [`LogLevel`] to a [`log::LevelFilter`].
28#[must_use]
29pub const fn to_log_level_filter(level: LogLevel) -> log::LevelFilter {
30    match level {
31        LogLevel::ALL | LogLevel::TRACE => log::LevelFilter::Trace,
32        LogLevel::DEBUG => log::LevelFilter::Debug,
33        LogLevel::VERBOSE | LogLevel::INFO => log::LevelFilter::Info,
34        LogLevel::WARN => log::LevelFilter::Warn,
35        LogLevel::ERROR | LogLevel::FATAL | LogLevel::CRITICAL => log::LevelFilter::Error,
36        LogLevel::NONE | LogLevel::DISABLED => log::LevelFilter::Off,
37    }
38}
39
40/// [`log::Log`] implementation that routes records into the logging ring buffer.
41#[derive(Debug, Clone, Copy)]
42pub struct LoggingFacade {
43    format: LogFormat,
44}
45
46impl LoggingFacade {
47    /// Create a `LoggingFacade` that formats output in the given format.
48    #[must_use]
49    pub const fn new(format: LogFormat) -> Self {
50        Self { format }
51    }
52}
53
54impl log::Log for LoggingFacade {
55    fn enabled(&self, metadata: &log::Metadata<'_>) -> bool {
56        map_log_level(metadata.level()).to_numeric() >= ENGINE.filter_level()
57    }
58
59    fn log(&self, record: &log::Record<'_>) {
60        if !self.enabled(record.metadata()) {
61            return;
62        }
63
64        let level = map_log_level(record.level());
65        let mut entry = Log::build(level, &record.args().to_string());
66        entry.component = std::borrow::Cow::Owned(record.target().to_string());
67        entry.format = self.format;
68
69        if let Some(file) = record.file() {
70            entry = entry.with("file", file);
71        }
72        if let Some(line) = record.line() {
73            entry = entry.with("line", line);
74        }
75        if let Some(module) = record.module_path() {
76            entry = entry.with("module", module);
77        }
78
79        entry.fire();
80    }
81
82    fn flush(&self) {
83        // The background flusher thread handles I/O.
84    }
85}

152    #[cfg_attr(miri, ignore)]
153    fn test_logging_facade_enabled() {
154        let logger = LoggingFacade::new(LogFormat::JSON);
155        // Reset filter to 0 (ALL) so everything is enabled
156        ENGINE.set_filter(0);
157
158        let metadata = log::MetadataBuilder::new().level(log::Level::Trace).build();
159        assert!(log::Log::enabled(&logger, &metadata));
160
161        let metadata = log::MetadataBuilder::new().level(log::Level::Error).build();
162        assert!(log::Log::enabled(&logger, &metadata));
163
164        // Test with a high filter: only ERROR should be enabled
165        ENGINE.set_filter(LogLevel::ERROR.to_numeric());
166        let trace_metadata = log::MetadataBuilder::new().level(log::Level::Trace).build();
167        assert!(!log::Log::enabled(&logger, &trace_metadata));
168
169        let error_metadata = log::MetadataBuilder::new().level(log::Level::Error).build();
170        assert!(log::Log::enabled(&logger, &error_metadata));
171
172        // Reset filter back
173        ENGINE.set_filter(0);
174    }
