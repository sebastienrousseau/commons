5//! Near-lock-free ingestion engine backed by a bounded ring buffer.
6//!
7//! The global [`ENGINE`] accepts [`LogEvent`]s via [`LockFreeEngine::ingest()`]
8//! using only atomic operations. A dedicated background thread drains events
9//! in batches of 64 and writes them through [`PlatformSink`](super::sink::PlatformSink).
10//!
11//! **The Mutex is never locked on the hot path.** It exists solely for
12//! `shutdown()` to join the flusher thread.
13
14use super::log_level::LogLevel;
15#[cfg(not(miri))]
16use super::sink::PlatformSink;
17use super::tui::TuiMetrics;
18#[cfg(not(miri))]
19use super::tui::spawn_tui_thread;
20use crossbeam_queue::ArrayQueue;
21use std::fmt;
22use std::sync::atomic::{AtomicBool, AtomicU8, Ordering};
23use std::sync::{Arc, LazyLock, Mutex};
24use std::thread;
25#[cfg(not(miri))]
26use std::time::Duration;

28/// Capacity of the lock-free ring buffer (number of log events).
29const RING_BUFFER_CAPACITY: usize = 65_536;
30
31/// Maximum number of events drained per flusher wake-up cycle.
32#[cfg(not(miri))]
33const MAX_DRAIN_BATCH_SIZE: usize = 64;
34
35/// A structured log event passed through the ring buffer.
36///
37/// The caller pays only for a `Log` move (~128-byte memcpy).
38/// Serialization happens on the flusher thread.
39#[derive(Debug, Clone)]
40pub struct LogEvent {
41    /// Severity level of this event.
42    pub level: LogLevel,
43    /// Numeric severity for fast level-gating comparisons.
44    pub level_num: u8,
45    /// Structured log data. Formatted on the flusher thread, not here.
46    pub log: super::log_entry::Log,
47}
48
49/// The near-lock-free ingestion engine.
50///
51/// Owns the ring buffer, flusher thread, and TUI metrics counters.
52/// Access the global instance via [`ENGINE`].
53pub struct LockFreeEngine {
54    /// Bounded ring buffer (lock-free push/pop via `crossbeam`).
55    queue: Arc<ArrayQueue<LogEvent>>,
56    /// Signals the flusher thread to drain and exit.
57    shutdown_flag: Arc<AtomicBool>,
58    /// Atomic counters consumed by the opt-in TUI dashboard.
59    metrics: Arc<TuiMetrics>,
60    /// Minimum severity level. Events below this are dropped at `ingest()`.
61    filter_level: AtomicU8,
62    /// Flusher thread handle for lock-free `unpark()`. No Mutex involved.
63    flusher_thread_handle: Option<thread::Thread>,
64    /// `JoinHandle` for `shutdown()` only. **Never locked on the hot path.**
65    flusher_join: Mutex<Option<thread::JoinHandle<()>>>,
66}
67
68impl fmt::Debug for LockFreeEngine {
69    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
70        f.debug_struct("LockFreeEngine")
71            .field("queue", &self.queue)
72            .field("shutdown_flag", &self.shutdown_flag)
73            .field("metrics", &self.metrics)
74            .field("filter_level", &self.filter_level)
75            .field(
76                "flusher_thread_handle",
77                &self.flusher_thread_handle.as_ref().map(thread::Thread::id),
78            )
79            .finish_non_exhaustive()
80    }
81}

83/// Global engine instance, lazily initialized on first access.
84pub static ENGINE: LazyLock<LockFreeEngine> =
85    LazyLock::new(|| LockFreeEngine::new(RING_BUFFER_CAPACITY));
86
87impl LockFreeEngine {
88    /// Create a new engine with the given buffer capacity and spawn the flusher.
89    ///
90    /// # Panics
91    ///
92    /// Panics if the OS cannot spawn the background flusher thread.
93    #[must_use]
94    pub fn new(capacity: usize) -> Self {
95        let queue = Arc::new(ArrayQueue::new(capacity));
96        let shutdown_flag = Arc::new(AtomicBool::new(false));
97        let metrics = Arc::new(TuiMetrics::default());
98        let filter_level = AtomicU8::new(0); // Default to ALL
99
100        // Under MIRI, skip spawning background threads to avoid
101        // "main thread terminated without waiting" errors.
102        #[cfg(not(miri))]
103        let flusher_handle = {
104            let flusher_queue = queue.clone();
105            let flusher_shutdown = shutdown_flag.clone();
106
107            // Spawn lightweight OS thread (Runtime Agnostic)
108            let handle = thread::Builder::new()
109                .name("logging-flusher".into())
110                .spawn(move || {
111                    use std::io::Write;
112                    let mut sink = PlatformSink::native();
113                    let mut fmt_buf = Vec::with_capacity(512);
114
115                    loop {
116                        let mut batch: [Option<LogEvent>; MAX_DRAIN_BATCH_SIZE] =
117                            std::array::from_fn(|_| None);
118                        let mut count = 0;
119                        while count < MAX_DRAIN_BATCH_SIZE {
120                            match flusher_queue.pop() {
121                                Some(event) => {
122                                    batch[count] = Some(event);
123                                    count += 1;
124                                }
125                                None => break,
126                            }
127                        }
128                        for event in batch.iter().flatten() {
129                            fmt_buf.clear();
130                            let _ = writeln!(fmt_buf, "{}", event.log);
131                            sink.emit(event.level.as_str(), &fmt_buf);
132                        }
133
134                        if flusher_shutdown.load(Ordering::Relaxed) && flusher_queue.is_empty() {
135                            break;
136                        }
137
138                        // Park briefly as fallback; real wakeup comes from unpark() in ingest().
139                        thread::park_timeout(Duration::from_millis(5));
140                    }
141                })
142                .expect("Failed to spawn logging-flusher background thread");
143
144            // Spawn the TUI dashboard thread if RLG_TUI=1
145            if std::env::var("RLG_TUI").is_ok_and(|v| v == "1") {
146                spawn_tui_thread(metrics.clone(), shutdown_flag.clone());
147            }
148
149            Some(handle)
150        };
151
152        #[cfg(miri)]
153        let flusher_handle: Option<thread::JoinHandle<()>> = None;
154
155        let flusher_thread_handle = flusher_handle.as_ref().map(|h| h.thread().clone());
156
157        Self {
158            queue,
159            shutdown_flag,
160            metrics,
161            filter_level,
162            flusher_thread_handle,
163            flusher_join: Mutex::new(flusher_handle),
164        }
165    }
166
167    /// Appends an event to the ring buffer.
168    ///
169    /// If the buffer is full, the oldest event is evicted to make room.
170    /// Dropped events are tracked via `TuiMetrics::dropped_events`.
171    pub fn ingest(&self, event: LogEvent) {
172        if event.level_num < self.filter_level.load(Ordering::Acquire) {
173            return;
174        }
175
176        self.metrics.inc_events();
177        self.metrics.inc_level(event.level);
178
179        if event.level_num >= LogLevel::ERROR.to_numeric() {
180            self.metrics.inc_errors();
181        }
182
183        // If the buffer is full, evict and retry with bounded retries.
184        if let Err(rejected) = self.queue.push(event) {
185            self.metrics.inc_dropped();
186            let mut to_push = rejected;
187            for _ in 0..3 {
188                let _ = self.queue.pop();
189                match self.queue.push(to_push) {
190                    Ok(()) => break,
191                    Err(e) => to_push = e,
192                }
193            }
194        }
195
196        // Wake the flusher thread -- no Mutex on the hot path.
197        if let Some(thread) = &self.flusher_thread_handle {
198            thread.unpark();
199        }
200    }
201
202    /// Sets the global log level filter.
203    pub fn set_filter(&self, level: u8) {
204        self.filter_level.store(level, Ordering::Release);
205    }
206
207    /// Returns the current global log level filter.
208    #[must_use]
209    pub fn filter_level(&self) -> u8 {
210        self.filter_level.load(Ordering::Relaxed)
211    }
212
213    /// Increments the format counter in the TUI metrics.
214    pub fn inc_format(&self, format: super::log_format::LogFormat) {
215        self.metrics.inc_format(format);
216    }
217
218    /// Increments the active span count in the TUI metrics.
219    pub fn inc_spans(&self) {
220        self.metrics.inc_spans();
221    }
222
223    /// Decrements the active span count in the TUI metrics.
224    pub fn dec_spans(&self) {
225        self.metrics.dec_spans();
226    }
227
228    /// Returns the current number of active spans.
229    #[must_use]
230    pub fn active_spans(&self) -> usize {
231        self.metrics.active_spans.load(Ordering::Relaxed)
232    }
233
234    /// Applies configuration settings to the engine.
235    ///
236    /// Sets the log level filter from the config. File sink construction
237    /// and rotation are handled by the flusher thread at startup via
238    /// [`PlatformSink::from_config`](super::sink::PlatformSink::from_config).
239    pub fn apply_config(&self, config: &super::log_config::LoggingConfig) {
240        self.set_filter(config.log_level.to_numeric());
241    }
242
243    /// Safely halts the background thread, flushing pending logs.
244    ///
245    /// Signals the flusher thread to stop and waits for it to finish
246    /// draining any remaining events from the queue.
247    pub fn shutdown(&self) {
248        self.shutdown_flag.store(true, Ordering::SeqCst);
249        // Wake the flusher so it can drain and exit.
250        if let Some(thread) = &self.flusher_thread_handle {
251            thread.unpark();
252        }
253        if let Ok(mut guard) = self.flusher_join.lock()
254            && let Some(handle) = guard.take()
255        {
256            let _ = handle.join();
257        }
258    }
259}

261/// Zero-Allocation Serializer Helper
262#[derive(Debug, Clone, Copy)]
263pub struct FastSerializer;
264
265impl FastSerializer {
266    /// Appends a u64 integer to a buffer using `itoa` without allocating a String.
267    pub fn append_u64(buf: &mut Vec<u8>, val: u64) {
268        let mut buffer = itoa::Buffer::new();
269        buf.extend_from_slice(buffer.format(val).as_bytes());
270    }
271
272    /// Appends an f64 float to a buffer using `ryu` without allocating a String.
273    pub fn append_f64(buf: &mut Vec<u8>, val: f64) {
274        let mut buffer = ryu::Buffer::new();
275        buf.extend_from_slice(buffer.format(val).as_bytes());
276    }
277}

485        // Capacity 2 so we can fill it and trigger eviction.
