7use super::log_config::LoggingConfigError;
8#[cfg(feature = "logging-miette")]
9use miette::Diagnostic;
10use std::fmt;
11use std::io;
12use thiserror::Error;

14/// Error variants for the logging pipeline.
15#[derive(Error, Debug)]
16#[cfg_attr(feature = "logging-miette", derive(Diagnostic))]
17pub enum LoggingError {
18    /// I/O error
19    #[error("I/O error: {0}")]
20    #[cfg_attr(
21        feature = "logging-miette",
22        diagnostic(
23            code(logging::io_error),
24            help("Ensure the log directory exists and is writable.")
25        )
26    )]
27    IoError(#[from] io::Error),
28
29    /// Configuration error
30    #[error("Configuration error: {0}")]
31    #[cfg_attr(
32        feature = "logging-miette",
33        diagnostic(
34            code(logging::config_error),
35            help("Check your configuration file or environment variables.")
36        )
37    )]
38    ConfigError(#[from] LoggingConfigError),
39
40    /// Log format parse error
41    #[error("Log format parse error: {0}")]
42    #[cfg_attr(
43        feature = "logging-miette",
44        diagnostic(
45            code(logging::format_parse_error),
46            help("Ensure the format string matches supported variants (JSON, OTLP, MCP, etc.).")
47        )
48    )]
49    FormatParseError(String),
50
51    /// Log level parse error
52    #[error("Log level parse error: {0}")]
53    #[cfg_attr(
54        feature = "logging-miette",
55        diagnostic(
56            code(logging::level_parse_error),
57            help("Supported levels: ALL, TRACE, DEBUG, INFO, WARN, ERROR, FATAL.")
58        )
59    )]
60    LevelParseError(String),
61
62    /// Unsupported log format
63    #[error("Unsupported log format: {0}")]
64    #[cfg_attr(
65        feature = "logging-miette",
66        diagnostic(
67            code(logging::unsupported_format),
68            help("Visit docs.rs/euxis-commons for a list of supported industry formats.")
69        )
70    )]
71    UnsupportedFormat(String),
72
73    /// Log formatting error
74    #[error("Log formatting error: {0}")]
75    #[cfg_attr(
76        feature = "logging-miette",
77        diagnostic(
78            code(logging::formatting_error),
79            help("This may happen if attributes contain non-serializable data.")
80        )
81    )]
82    FormattingError(String),
83
84    /// Log rotation error
85    #[error("Log rotation error: {0}")]
86    #[cfg_attr(
87        feature = "logging-miette",
88        diagnostic(
89            code(logging::rotation_error),
90            help("Ensure the logger has permission to rename or delete old log files.")
91        )
92    )]
93    RotationError(String),
94
95    /// Network error
96    #[error("Network error: {0}")]
97    #[cfg_attr(
98        feature = "logging-miette",
99        diagnostic(
100            code(logging::network_error),
101            help("Check your network connection or the OTLP collector endpoint.")
102        )
103    )]
104    NetworkError(String),
105
106    /// `DateTime` parse error
107    #[error("DateTime parse error: {0}")]
108    #[cfg_attr(
109        feature = "logging-miette",
110        diagnostic(
111            code(logging::datetime_parse_error),
112            help("Expects RFC 3339 / ISO 8601 timestamps.")
113        )
114    )]
115    DateTimeParseError(String),
116
117    /// Custom error
118    #[error("{0}")]
119    #[cfg_attr(feature = "logging-miette", diagnostic(code(logging::custom_error)))]
120    Custom(String),
121
122    /// Native OS sink failure
123    #[error("Native OS sink failure: {0}")]
124    #[cfg_attr(
125        feature = "logging-miette",
126        diagnostic(
127            code(logging::native_sink_failure),
128            help(
129                "Check if systemd-journald is running (Linux). Ensure RLG_FALLBACK_STDOUT is set to bypass native hooks."
130            )
131        )
132    )]
133    NativeSinkError(String),
134}
135
136impl From<crate::error::CommonError> for LoggingError {
137    fn from(err: crate::error::CommonError) -> Self {
138        Self::Custom(err.to_string())
139    }
140}
141
142impl LoggingError {
143    /// Create a custom error with the given message.
144    #[must_use]
145    pub fn custom<T: fmt::Display>(msg: T) -> Self {
146        Self::Custom(msg.to_string())
147    }
148}
149
150/// Convenience alias: `Result<T, LoggingError>`.
151pub type LoggingResult<T> = Result<T, LoggingError>;

153#[cfg(test)]
154mod tests {
155    use super::*;
156
157    #[test]
158    fn test_error_display() {
159        let err = LoggingError::FormatParseError("Invalid format".to_string());
160        assert_eq!(err.to_string(), "Log format parse error: Invalid format");
161    }
162
163    #[test]
164    fn test_custom_error() {
165        let err = LoggingError::custom("Custom error message");
166        assert_eq!(err.to_string(), "Custom error message");
167    }
168
169    #[test]
170    fn test_common_error_conversion() {
171        let common_err = crate::error::CommonError::custom("test");
172        let logging_err: LoggingError = common_err.into();
173        assert!(matches!(logging_err, LoggingError::Custom(_)));
174        assert!(logging_err.to_string().contains("test"));
175    }
176
177    #[test]
178    fn test_io_error_variant() {
179        let io_err = io::Error::new(io::ErrorKind::NotFound, "file missing");
180        let logging_err: LoggingError = io_err.into();
181        assert!(matches!(logging_err, LoggingError::IoError(_)));
182        assert!(logging_err.to_string().contains("file missing"));
183    }
184
185    #[test]
186    fn test_all_variant_display() {
187        let variants: Vec<LoggingError> = vec![
188            LoggingError::IoError(io::Error::other("test")),
189            LoggingError::ConfigError(LoggingConfigError::ValidationError("v".into())),
190            LoggingError::FormatParseError("f".into()),
191            LoggingError::LevelParseError("l".into()),
192            LoggingError::UnsupportedFormat("u".into()),
193            LoggingError::FormattingError("fm".into()),
194            LoggingError::RotationError("r".into()),
195            LoggingError::NetworkError("n".into()),
196            LoggingError::DateTimeParseError("d".into()),
197            LoggingError::Custom("c".into()),
198            LoggingError::NativeSinkError("ns".into()),
199        ];
200        for err in &variants {
201            assert!(!format!("{err:?}").is_empty());
202        }
203    }
204
205    #[test]
206    fn test_logging_result_ok() {
207        let r: LoggingResult<i32> = Ok(42);
208        assert!(matches!(r, Ok(42)));
209    }
210
211    #[test]
212    fn test_logging_result_err() {
213        let r: LoggingResult<i32> = Err(LoggingError::custom("fail"));
214        assert!(r.is_err());
215    }
216}
